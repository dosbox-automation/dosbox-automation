# Build libdecor core + decoration plugin for bundled Wayland CSD.
# Produces standalone shared libraries loaded at runtime by SDL3's
# libdecor integration, NOT linked into the dosbox binary.
#
# Build requirements: wayland-scanner, wayland-protocols, wayland-client,
# wayland-cursor. The cairo plugin (Phase 1 proof) additionally needs
# pangocairo and cairo.

cmake_minimum_required(VERSION 3.20)

find_program(WAYLAND_SCANNER wayland-scanner)
if(NOT WAYLAND_SCANNER)
  message(STATUS "wayland-decor: wayland-scanner not found, skipping")
  return()
endif()

find_package(PkgConfig REQUIRED)

pkg_check_modules(WAYLAND_CLIENT IMPORTED_TARGET wayland-client)
if(NOT WAYLAND_CLIENT_FOUND)
  message(STATUS "wayland-decor: wayland-client not found, skipping")
  return()
endif()

pkg_check_modules(WAYLAND_CURSOR IMPORTED_TARGET wayland-cursor)
if(NOT WAYLAND_CURSOR_FOUND)
  message(STATUS "wayland-decor: wayland-cursor not found, skipping")
  return()
endif()

pkg_get_variable(WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
if(NOT WAYLAND_PROTOCOLS_DIR)
  message(STATUS "wayland-decor: wayland-protocols not found, skipping")
  return()
endif()

message(STATUS "wayland-decor: building libdecor + decoration plugin")

set(LIBDECOR_SRC "${CMAKE_SOURCE_DIR}/vendor/libdecor")
set(PROTO_OUT "${CMAKE_CURRENT_BINARY_DIR}/protocols")
file(MAKE_DIRECTORY ${PROTO_OUT})

# --- Wayland protocol code generation ---

function(generate_wayland_protocol stability name)
  set(extra_args ${ARGN})
  if(stability STREQUAL "unstable")
    list(GET extra_args 0 version)
    set(xml_path "${WAYLAND_PROTOCOLS_DIR}/${stability}/${name}/${name}-unstable-v${version}.xml")
  else()
    set(xml_path "${WAYLAND_PROTOCOLS_DIR}/${stability}/${name}/${name}.xml")
  endif()

  set(header "${PROTO_OUT}/${name}-client-protocol.h")
  set(source "${PROTO_OUT}/${name}-protocol.c")

  add_custom_command(
    OUTPUT ${header}
    COMMAND ${WAYLAND_SCANNER} client-header ${xml_path} ${header}
    DEPENDS ${xml_path}
    VERBATIM
  )
  add_custom_command(
    OUTPUT ${source}
    COMMAND ${WAYLAND_SCANNER} private-code ${xml_path} ${source}
    DEPENDS ${xml_path}
    VERBATIM
  )

  set(PROTO_SOURCES ${PROTO_SOURCES} ${header} ${source} PARENT_SCOPE)
endfunction()

set(PROTO_SOURCES "")
generate_wayland_protocol(stable xdg-shell)
generate_wayland_protocol(unstable xdg-decoration 1)

# --- config.h for libdecor ---

set(LIBDECOR_CONFIG_H "${CMAKE_CURRENT_BINARY_DIR}/config.h")

include(CheckFunctionExists)

check_function_exists(posix_fallocate HAVE_POSIX_FALLOCATE)
check_function_exists(memfd_create HAVE_MEMFD_CREATE)
check_function_exists(gettid HAVE_GETTID)

pkg_check_modules(DBUS dbus-1)

file(WRITE ${LIBDECOR_CONFIG_H}
"/* generated by CMake for libdecor */
#define VERSION \"0.2.2\"
#define LIBDECOR_PLUGIN_DIR \"\"
#define LIBDECOR_PLUGIN_API_VERSION 1
")

if(DBUS_FOUND)
  file(APPEND ${LIBDECOR_CONFIG_H} "#define HAS_DBUS 1\n")
endif()
if(HAVE_MKOSTEMP)
  file(APPEND ${LIBDECOR_CONFIG_H} "#define HAVE_MKOSTEMP 1\n")
endif()
if(HAVE_POSIX_FALLOCATE)
  file(APPEND ${LIBDECOR_CONFIG_H} "#define HAVE_POSIX_FALLOCATE 1\n")
endif()
if(HAVE_MEMFD_CREATE)
  file(APPEND ${LIBDECOR_CONFIG_H} "#define HAVE_MEMFD_CREATE 1\n")
endif()
if(HAVE_GETTID)
  file(APPEND ${LIBDECOR_CONFIG_H} "#define HAVE_GETTID 1\n")
endif()

# --- desktop-settings + os-compatibility helpers ---
#
# Meson builds these as static libs linked into the core AND each
# plugin separately; hidden visibility keeps the copies private, so
# the plugin cannot resolve them from the shared library.

add_library(decor-helpers STATIC
  ${LIBDECOR_SRC}/src/desktop-settings.c
  ${LIBDECOR_SRC}/src/os-compatibility.c
)
set_target_properties(decor-helpers PROPERTIES
  C_STANDARD 99
  POSITION_INDEPENDENT_CODE ON
  C_VISIBILITY_PRESET hidden
)
target_compile_definitions(decor-helpers PRIVATE _GNU_SOURCE)
target_compile_options(decor-helpers PRIVATE -w)
target_include_directories(decor-helpers PRIVATE
  ${CMAKE_CURRENT_BINARY_DIR}
  ${LIBDECOR_SRC}/src
)
if(DBUS_FOUND)
  target_include_directories(decor-helpers PRIVATE ${DBUS_INCLUDE_DIRS})
  target_link_libraries(decor-helpers PRIVATE ${DBUS_LIBRARIES})
endif()

# --- libdecor core (shared library) ---

add_library(decor-0 SHARED
  ${LIBDECOR_SRC}/src/libdecor.c
  ${LIBDECOR_SRC}/src/libdecor-fallback.c
  ${PROTO_SOURCES}
)

set_target_properties(decor-0 PROPERTIES
  OUTPUT_NAME "decor-0"
  VERSION 0.2.2
  SOVERSION 0
  C_STANDARD 99
  POSITION_INDEPENDENT_CODE ON
  C_VISIBILITY_PRESET hidden
)

target_compile_definitions(decor-0 PRIVATE _GNU_SOURCE)
target_compile_options(decor-0 PRIVATE -w)

target_include_directories(decor-0 PRIVATE
  ${CMAKE_CURRENT_BINARY_DIR}
  ${PROTO_OUT}
  ${LIBDECOR_SRC}
  ${LIBDECOR_SRC}/src
)

target_link_libraries(decor-0 PRIVATE
  decor-helpers
  PkgConfig::WAYLAND_CLIENT
  ${CMAKE_DL_LIBS}
)
